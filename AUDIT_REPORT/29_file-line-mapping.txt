     export async function uploadOfferImageController(req: Request, res: Response, next: NextFunction): Promise<void> {
       const { id } = req.params;
       if (!id) { next(new AppError(400, 'ID de l\'offre requis.')); return; }
       if (!req.file) { next(new AppError(400, 'Aucune image fournie.')); return; }
       try {
         const offer = await setOfferImage(String(id), req.file.filename);
         res.json({ offer });
       } catch (err) {
         // On failure (offer not found or otherwise), remove the freshly uploaded file...
         await removeUploadedFile(`${OFFERS_PUBLIC_PREFIX}/${req.file.filename}`);
         if (err instanceof Error && err.message === 'OFFER_NOT_FOUND') {
           next(new AppError(404, 'Offre non trouvée.')); return;
         }
         next(err);
       }
     }

9. SERVICE: ORPHAN CLEANUP
   File: backend/src/services/offer.service.ts
   Lines: 51-59
   Code:
     // Attaches a photo (already written to disk by multer) to an offer.
     // The previous photo, if any, is removed best-effort after the DB update.
     export const setOfferImage = async (id: string, filename: string) => {
       const existing = await prisma.partnerOffer.findUnique({ where: { id }, select: { imageUrl: true } })
       if (!existing) throw new Error('OFFER_NOT_FOUND')
       const imageUrl = `${OFFERS_PUBLIC_PREFIX}/${filename}`
       const updated = await prisma.partnerOffer.update({ where: { id }, data: { imageUrl } })
       if (existing.imageUrl) await removeUploadedFile(existing.imageUrl)
       return updated
     }

      export function requireAuth(req: Request, res: Response, next: NextFunction): void {
        const header = req.headers.authorization;
        if (!header?.startsWith('Bearer ')) {
          next(new AppError(401, 'Token manquant.'));
          return;
        }
        const token = header.slice(7);
        try {
          const payload = verifyAccessToken(token);
          req.user = {
            id: payload.sub,
            role: payload.role,
            ...(payload.companyId ? { companyId: payload.companyId } : {}),
          };
          next();
        } catch {
          next(new AppError(401, 'Token invalide ou expiré.'));
        }
      }

14. MIDDLEWARE: AUTHORIZATION (ADMIN CHECK)

      export async function requireAdmin(req: Request, res: Response, next: NextFunction): Promise<void> {
        // The token says ADMIN, but role/status can change mid-session...
        if (req.user?.role !== 'ADMIN') {
          next(new AppError(403, 'Accès réservé aux administrateurs.'));
          return;
        }
        try {
          const user = await prisma.user.findFirst({
            where: { id: req.user.id, deletedAt: null },
            select: { role: true },
          });
          if (!user || user.role !== 'ADMIN') {
            next(new AppError(403, 'Accès réservé aux administrateurs.'));
            return;
          }
          next();
        } catch (err) {
          next(err);
        }
      }


VULNERABILITY LOCATIONS
=======================

MISSING: Magic Byte Validation
   File: backend/src/lib/upload.ts
   Lines: 34-37
   Issue: fileFilter only checks file.mimetype (client-controlled header)
   What's Missing: No validation of actual file binary signatures
   Example Attack:
     Upload PHP code with Content-Type: image/jpeg
     Server accepts because mimetype = 'image/jpeg'
     No magic byte check performed
   Risk Level: MEDIUM (mitigated by static serving)
   Recommendation: Add file-type npm package for magic byte validation

IMPLICIT: Express JSON Limit
   File: backend/src/app.ts
   Line: 68
   Code: app.use(express.json());
   Issue: No explicit limit parameter
   Default: 100kb (Express 5.x)
   Risk Level: LOW (safe default, but implicit)
   Recommendation: app.use(express.json({ limit: '1mb' }));

IMPLICIT: Multipart Field Limits
   File: backend/src/lib/upload.ts
   Lines: 31-37
   Issue: multer config only specifies fileSize, not fieldSize or fields
   Risk Level: LOW (.single('image') rejects extra fields)
   Recommendation:
     limits: {
       fileSize: 2 * 1024 * 1024,
       files: 1,
       fields: 1,
       fieldSize: 1 * 1024,
     }
